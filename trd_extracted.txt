1. Technical Overview
This document specifies the complete architecture, component design, state management, data models, simulation logic, gamification implementation, audio pipeline, and quality standards for the "Subtraction within 100 Using the Relationship Between Addition and Subtraction" interactive lesson module (Grade 2 Math) within Intellia SG's curriculum platform.

The module is a React 18 application (Vite + JSX), structured identically to the reference repository https://github.com/dsamyak/numberbound and styled to exactly match https://numberbound.vercel.app/. It will be embedded in WordPress at https://intelliasg.com/courses/grade-2-math/.

5.1 Question Schema (TypeScript-style)

interface Question {
  id:            string;        // e.g. "Q1_007", "Q9_042"
  type:          QuestionType;  // 10 types (see below)
  world:         number;        // 0–9
  difficulty:    1 | 2 | 3;     // 1=easy(≤50), 2=med(≤80), 3=hard(≤100)
  whole:         number;        // Minuend (total)
  part1:         number;        // Known subtrahend
  part2:         number;        // Difference (answer)
  missingSlot:   'whole' | 'part1' | 'part2';
  questionText:  string;
  visual:        'triangle' | 'blocks' | 'barModel' | 'sentence' | 'trueFalse' | 'picture';
  counterEmoji?: string;
  options?:      (string|number)[];  // Always 4 MCQ options
  hint1:         string;
  hint2:         string;
  explanation:   string;
  characterName?: string;       // English names only
  objectName?:   string;
  correctAnswer: number | string;
}

type QuestionType = 'fact_family_find' | 'fill_blank' | 'inverse_check' | 'word_problem'
                  | 'complete_family' | 'missing_addend' | 'bar_model'
                  | 'true_false_bond' | 'regrouping' | 'how_many_more';

8. Randomisation Engine
8.1 Fisher-Yates Shuffle

// src/utils/shuffle.js
export function shuffleArray(array) {
  const arr = [...array];
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

export function generateSessionQuestions(bank) {
  // Group by type (10 types), pick 10 per type, final shuffle of 100
  const byType = bank.reduce((acc, q) => {
    if (!acc[q.type]) acc[q.type] = [];
    acc[q.type].push(q);
    return acc;
  }, {});
  const selected = Object.values(byType)
    .flatMap(qs => shuffleArray(qs).slice(0, 10));
  return shuffleArray(selected); // 100 fully randomised
}


8.2 MCQ Distractor Generation

// src/utils/scoring.js
export function generateDistractors(correct, min = 0, max = 100, count = 3) {
  const distractors = new Set();
  let attempts = 0;
  // Common error patterns for subtraction within 100:
  const errorPatterns = [correct + 1, correct - 1, correct + 10, correct - 10];
  errorPatterns.forEach(d => {
    if (d >= min && d <= max && d !== correct && distractors.size < count)
      distractors.add(d);
  });
  // Fill remaining slots with nearby random values
  while (distractors.size < count && attempts < 50) {
    const offset = (Math.random() > 0.5 ? 1 : -1) * (Math.ceil(Math.random() * 5));
    const d = correct + offset;
    if (d >= min && d <= max && d !== correct) distractors.add(d);
    attempts++;
  }
  return shuffleArray([correct, ...distractors]);
}

9.3 Frontend Audio Engine (src/utils/audio.js)

import { audioMap } from './audioMap';

const VOICE_ID = 'Xb7hH8MSUJpSbSDYk0k2';
const MODEL_ID = 'eleven_multilingual_v2';
const elevenLabsCache = {};

export async function getAudioUrl(text, style = 'statement') {
  // 1. Check pre-generated map first
  if (audioMap[text]) return audioMap[text];

  // 2. Check runtime cache
  if (elevenLabsCache[text]) return elevenLabsCache[text];

  // 3. Dynamic generation (no key = silent skip)
  const apiKey = import.meta.env.VITE_ELEVENLABS_API_KEY;
  if (!apiKey) return null;

  const voiceSettings = {
    statement:    { stability:0.65, similarity_boost:0.80, style:0.10 },
    question:     { stability:0.55, similarity_boost:0.85, style:0.25 },
    encouragement:{ stability:0.45, similarity_boost:0.90, style:0.45 },
    emphasis:     { stability:0.80, similarity_boost:0.80, style:0.05 },
    thinking:     { stability:0.70, similarity_boost:0.75, style:0.20 },
    celebration:  { stability:0.35, similarity_boost:0.95, style:0.60 },
  }[style] ?? { stability:0.65, similarity_boost:0.80, style:0.10 };

  const res = await fetch(`https://api.elevenlabs.io/v1/text-to-speech/${VOICE_ID}`, {
    method: 'POST',
    headers: { 'xi-api-key': apiKey, 'Content-Type': 'application/json' },
    body: JSON.stringify({ text, model_id: MODEL_ID, voice_settings: voiceSettings })
  });

  const blob = await res.blob();
  const url = URL.createObjectURL(blob);
  elevenLabsCache[text] = url;
  return url;
}

export async function speak(text, style = 'statement') {
  const url = await getAudioUrl(text, style);
  if (!url) return;
  const audio = new Audio(url);
  return new Promise((res) => { audio.onended = res; audio.play(); });
}

export async function narrate(segments) {
  // Eager preload next segment while current plays
  for (let i = 0; i < segments.length; i++) {
    if (i + 1 < segments.length) getAudioUrl(segments[i+1].text, segments[i+1].style);
    await speak(segments[i].text, segments[i].style);
  }
}

10.1 XP Calculation (src/utils/scoring.js)

const XP_TABLE = {
  correctFirstTry: 10,
  correctSecondTry: 7,

export function calcXP(attemptNumber, hintsUsed, streak) {
  const base = attemptNumber === 1 ? 10 : hintsUsed > 0 ? 5 : 7;
  const streakBonus = streak >= 5 ? 5 : 0; // Bonus at 5+ streak
  return base + streakBonus;
}

export function calcStars(correct, total = 10) {
  if (correct >= 9) return 3; // 9–10 correct = 3 stars
  if (correct >= 7) return 2; // 7–8 correct = 2 stars
  if (correct >= 5) return 1; // 5–6 correct = 1 star
  return 0;                   // < 5 correct = retry required
}

export function canUnlockWorld(worldScore) {
  return worldScore !== null && worldScore >= 5;
}

13. Session Persistence (localStorage)

// src/utils/sessionStorage.js
const SESSION_KEY = 'intellia_sub100_v1';
const SESSION_TTL = 86400000; // 24 hours in ms

export function saveSession(state) {
  localStorage.setItem(SESSION_KEY, JSON.stringify({
    phase:           state.phase,
    currentQuestion: state.currentQuestion,
    currentWorld:    state.currentWorld,
    worldScores:     state.worldScores,
    xp:              state.xp,
    badges:          state.badges,
    phaseComplete:   state.phaseComplete,
    simStationsComplete: state.simStationsComplete,
    timestamp:       Date.now(),
  }));
}

export function loadSession() {
  try {
    const raw = localStorage.getItem(SESSION_KEY);
    if (!raw) return null;
    const saved = JSON.parse(raw);
    if (Date.now() - saved.timestamp > SESSION_TTL) {
      localStorage.removeItem(SESSION_KEY);
      return null;
    }
    return saved;
  } catch { return null; }
}

export function clearSession() {
  localStorage.removeItem(SESSION_KEY);
}
